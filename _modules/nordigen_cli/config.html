class BaseConfigManager(ABC):
    def __init__(self):
        self.default_config = {}
        self.user_config = {}
        self.command_config = {}
        self.profile = None
        self.settings = None
        self.command_args = {}
        self.is_valid = False
        self.validation_errors = []
        self.merged_config = {}

    def _get_default_settings(self):
        return {
            "verbose": 1,
            "output_format": "rich",
        }

    def _get_default_profile(self):
        return {
            "api": "https://bankaccountdata.gocardless.xcom/api/v2",
            "secret_id": None,
            "secret_key": None,
        }

        return {
            "profile": "default",
            "settings": self._get_default_settings(),
            "profiles": {
                "default": self._get_default_profile(),
            },
        }

    def _get_default_user_config(self):
        return self._get_default_config()

class ConfigManager(BaseConfigManager):
    def __init__(self, *args, **kwargs):
        super().__init__()
        logger.debug(f"ConfigManager args: {args})")
        for k, v in kwargs.items():
            logger.debug(f"{k}: {v}")

        # special handling of config_dir
        self.config_dir = kwargs.pop(
            "config_dir", SettingsConfig.model_fields["config_dir"].default
        )

        # dict that is put into default profile
        self.default_config = self._get_default_config()

        # was config file specifically overridden?
        self.config_file = kwargs.pop("config_file", self.config_dir / "config.json")

        self.user_config = self._get_user_config()
        self.merged_config = recursive_merge(self.default_config, self.user_config)

        profile_name = kwargs.pop("profile", self.merged_config["profile"])
        profile = self._get_profile(profile_name)
        self.profile = profile

        if not profile:
            raise ValueError(f"Profile not found: {profile_name}")

        self.kwargs = kwargs

        self.command_args = {
            k: v
            for k, v in kwargs.items()
            if v and k in SettingsRuntime.model_fields.keys()
        }

        sources = [self.command_args, profile, self.merged_config["settings"]]

        self.args = {
            k: next((item[k] for item in sources if item.get(k) is not None), None)
            for k in SettingsRuntime.model_fields.keys()
        }

        self.display = SettingsDisplay(
            **{
                k: v
                for k, v in self.args.items()
                if v and k in SettingsDisplay.model_fields.keys()
            }
        )
        try:
            self.settings = SettingsRuntime(**self.args)
            self.is_valid = True
            logger.trace(self.settings)
        except ValidationError as e:
            self.validation_errors.append(e)

    def _get_user_config(self):
        if self.config_file.exists():
            return self._load_user_config()
        else:
            self.validation_errors.append(
                ValueError(f"User config file not found: {self.config_file}")
            )
            return {}

    def _get_profile(self, profile_name):
        profile = self.merged_config["profiles"].get(profile_name)
        if not profile:
            self.validation_errors.append(
                ValueError(f"Profile not found: {profile_name}")
            )
        return profile

    def _load_user_config(self):
        if not self.config_file.exists():
            logger.debug("No user config file exists - run init first")
        with open(self.config_file) as f:
            return json.load(f)

    def _get_command_settings(self, **kwargs):
        return {
            k: v for k, v in kwargs.items() if v and (k in self._get_default_settings())
        }

    def _get_command_overrides(self, **kwargs):
        return {
            k: v for k, v in kwargs.items() if v and (k in self._get_default_profile())
        }

    def _make_user_config(self):
        if not self.config_file.parent.exists():
            logger.debug(f"Creating local config directory: {self.config_file.parent}")
            self.config_file.parent.mkdir(exist_ok=True)
        if not self.config_file.exists():
            logger.debug(f"Creating user config file from defaults: {self.config_file}")
            with open(self.config_file, "w") as f:
                json.dump(self._get_default_user_config(), f, indent=4)

    def copy(self):
        new_config = ConfigManager()
        new_config.global_config = copy.deepcopy(self.global_config)
