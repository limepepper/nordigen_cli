def show_config_report(config: BaseConfigManager) -> None:
    """
    show the configuration error

    """
    logger.debug("showing configuration error")
    # print(ctx.parent.params)
    if config.is_valid and config.validation_errors == []:
        print("config is [bold green]valid![/] All settings are correct")
    elif config.is_valid:
        print("config is [bold yellow]valid![/] But there are some warnings. see below")
    else:
        print("config is [bold red]invalid![/bold red] See below for details")

    # print("[bold red]alert![/bold red] Something happened")

    print("merged settings")
    console.print(config.merged_config["settings"])

    print("profile settings")
    console.print(config.profile)

    print("command_args settings")
    console.print(config.command_args)

    # print("self.config settings")
    # console.print(config.merged_config)
    #
    # print("User config settings")
    # console.print(config.user_config)
    #
    # print("Overridden settings from cli options")
    # console.print(Pretty(config.command_args))
    #
    # print("Computed config settings")
    # console.print(Pretty(config.command_args))
    #
    print("Settings")
    console.print(Pretty(config.settings))

    for error in config.validation_errors:
        if isinstance(error, ValidationError):
            print("Validation Errors")
            for e in error.errors():
                pprint(e)

            inspect(error)
